const EXP_TABLE = new Array(512);
const LOG_TABLE = new Array(256);
(function initGF(){
  for (let i=0;i<8;i++) EXP_TABLE[i] = 1 << i;
  for (let i=8;i<256;i++) EXP_TABLE[i] = EXP_TABLE[i-4] ^ EXP_TABLE[i-5] ^ EXP_TABLE[i-6] ^ EXP_TABLE[i-8];
  for (let i=0;i<255;i++) LOG_TABLE[EXP_TABLE[i]] = i;
  for (let i=255;i<512;i++) EXP_TABLE[i] = EXP_TABLE[i-255];
})();
function gexp(n){ while(n<0) n += 255; return EXP_TABLE[n % 255]; }
function glog(n){ if(n < 1) throw new Error('glog('+n+')'); return LOG_TABLE[n]; }
function gfMul(a,b){ if(a===0 || b===0) return 0; return gexp(glog(a)+glog(b)); }
function polyMul(p,q){
  const out = new Array(p.length + q.length - 1).fill(0);
  for(let i=0;i<p.length;i++){
    for(let j=0;j<q.length;j++){
      out[i+j] ^= gfMul(p[i], q[j]);
    }
  }
  return out;
}
const genPolyCache = {};
function rsGeneratorPoly(ecCount){
  if(genPolyCache[ecCount]) return genPolyCache[ecCount].slice();
  let poly = [1];
  for(let i=0;i<ecCount;i++) poly = polyMul(poly, [1, gexp(i)]);
  genPolyCache[ecCount] = poly.slice();
  return poly;
}
function rsEncode(data, ecCount){
  const gen = rsGeneratorPoly(ecCount);
  const res = data.slice().concat(new Array(ecCount).fill(0));
  for(let i=0;i<data.length;i++){
    const coef = res[i];
    if(coef !== 0){
      for(let j=0;j<gen.length;j++) res[i+j] ^= gfMul(gen[j], coef);
    }
  }
  return res.slice(data.length);
}

function lengthBits(mode, version){
  if(version < 10){ if(mode===MODE.NUMERIC) return 10; if(mode===MODE.ALPHA) return 9; if(mode===MODE.BYTE) return 8; }
  else if(version < 27){ if(mode===MODE.NUMERIC) return 12; if(mode===MODE.ALPHA) return 11; if(mode===MODE.BYTE) return 16; }
  else { if(mode===MODE.NUMERIC) return 14; if(mode===MODE.ALPHA) return 13; if(mode===MODE.BYTE) return 16; }
  throw new Error('bad mode');
}
