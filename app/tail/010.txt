function generate(){
  els.errorBox.innerHTML = '';
  try{
    const payload = els.payload.value;
    const qr = buildQR(payload, {mode:els.mode.value, ecc:els.ecc.value, version:els.version.value, mask:els.mask.value, padMode:els.padMode.value});
    state.qr = qr; state.selection = null; state.edits = new Map();
    if(state.target && state.target.size!==qr.size) state.target=null;
    if(state.targetImage) mapTargetImage(state.targetImage);
    renderAll();
  }catch(e){
    state.qr = null;
    els.errorBox.innerHTML = `<div class="error">${esc(e.message || e)}</div>`;
    els.status.innerHTML = ''; els.charBar.innerHTML=''; els.blockGrid.innerHTML=''; els.details.textContent='';
    drawBlank();
  }
}
