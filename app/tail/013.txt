ata:new Set(), ecc:new Set(), block:new Set(), codeword:new Set(), module:null};
  if(!q || !sel) return sets;
  function addStream(set, idx){ (q.modulePositionsByStream[idx]||[]).forEach(p=>set.add(key(p.r,p.c))); }
  if(sel.type==='char'){
    const info=q.charInfos[sel.index];
    if(info){
      info.exactModules.forEach(p=>sets.exact.add(key(p.r,p.c)));
      info.dataStreamIndices.forEach(idx=>addStream(sets.data, idx));
      info.eccStreamIndices.forEach(idx=>addStream(sets.ecc, idx));
    }
  } else if(sel.type==='block'){
    const b=q.blocks.find(x=>x.id===sel.id);
    if(b) b.modules.forEach(p=>sets.block.add(key(p.r,p.c)));
  } else if(sel.type==='module'){
    sets.module = key(sel.r,sel.c);
    const m=q.meta[sel.r][sel.c];
    if(m && m.kind==='codeword') addStream(sets.codeword, m.streamIndex);
    if(m && typeof m.blockId==='number') (q.modulePositionsByBlock[m.blockId]||[]).forEach(p=>sets.block.add(key(p.r,p.c)));
  }
  return sets;
}
