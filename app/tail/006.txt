function baselineDark(r,c){ return !!state.qr.modules[r][c]; }
function currentDark(r,c){ const k=key(r,c); return state.edits.has(k) ? state.edits.get(k) : baselineDark(r,c); }
function setOverride(r,c,dark){
  const q=state.qr; if(!q) return false;
  const m=q.meta[r][c];
  if(!m || m.kind==='function') return false;
  const k=key(r,c), b=baselineDark(r,c);
  if(!!dark===b) state.edits.delete(k); else state.edits.set(k,!!dark);
  return true;
}
function analysisOfEdits(){
  const q=state.qr;
  const byBlock=new Map(), streams=new Set(); let remainder=0, ignored=0;
  if(!q) return {streams,byBlock,remainder,ignored,changedModules:0};
  for(const [k,val] of state.edits){
    const [r,c]=k.split(',').map(Number), m=q.meta[r]?.[c];
    if(!m){ignored++;continue;}
    if(m.kind==='codeword') streams.add(m.streamIndex);
    else if(m.kind==='remainder') remainder++;
    else ignored++;
  }
  for(const idx of streams){
    const cw=q.stream[idx]; if(!cw) continue;
    if(!byBlock.has(cw.blockId)) byBlock.set(cw.blockId,{all:new Set(),data:new Set(),ecc:new Set()});
    const x=byBlock.get(cw.blockId); x.all.add(idx); x[cw.kind].add(idx);
  }
  return {streams,byBlock,remainder,ignored,changedModules:state.edits.size};
}
