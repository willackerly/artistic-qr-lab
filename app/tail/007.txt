function targetCostForModule(r,c){
  if(!state.target || !state.target.active?.[r]?.[c]) return 0.7;
  const desired=state.target.dark[r][c], weight=state.target.weight[r][c] ?? 1;
  const edited=currentDark(r,c), base=baselineDark(r,c);
  if(edited===desired && base!==desired) return 1 + 4*weight;
  return .2 + weight*.3;
}
function repairCodewordCost(streamIndex){
  const q=state.qr; let cost=0, count=0;
  for(const p of (q.modulePositionsByStream[streamIndex]||[])){
    if(state.edits.has(key(p.r,p.c))){ cost += targetCostForModule(p.r,p.c); count++; }
  }
  return count ? cost : Infinity;
}
