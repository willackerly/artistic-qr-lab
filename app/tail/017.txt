function moduleFromEvent(ev){
  if(!state.qr || !state.geom) return null;
  const rect=els.canvas.getBoundingClientRect(); const scaleX=els.canvas.width/rect.width, scaleY=els.canvas.height/rect.height;
  const x=(ev.clientX-rect.left)*scaleX, y=(ev.clientY-rect.top)*scaleY;
  const {cell,ox,oy,margin}=state.geom;
  const col=Math.floor((x-ox)/cell)-margin; const r=Math.floor((y-oy)/cell)-margin;
  if(r<0 || col<0 || r>=state.qr.size || col>=state.qr.size) return null;
  return {r,c:col};
}
