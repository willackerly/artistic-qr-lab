function targetGridEmpty(n){ return {size:n,dark:Array.from({length:n},()=>Array(n).fill(false)),active:Array.from({length:n},()=>Array(n).fill(false)),weight:Array.from({length:n},()=>Array(n).fill(1)),label:''}; }
function makeSmileyTarget(){
  const q=state.qr; if(!q) return;
  const n=q.size,t=targetGridEmpty(n), cx=(n-1)/2,cy=(n-1)/2,R=n*.29, thick=Math.max(1,n*.026);
  for(let r=0;r<n;r++) for(let c=0;c<n;c++){
    const x=c-cx,y=r-cy,d=Math.hypot(x,y); const inBox=Math.abs(x)<=R*1.12 && Math.abs(y)<=R*1.12;
    if(!inBox) continue; t.active[r][c]=true;
    const ring=Math.abs(d-R)<thick;
    const eyeR=Math.max(1,n*.035); const ey=-R*.26, ex=R*.34;
    const eye=(Math.hypot(x-ex,y-ey)<eyeR)||(Math.hypot(x+ex,y-ey)<eyeR);
    const mouthR=R*.56; const md=Math.abs(Math.hypot(x,y+R*.02)-mouthR)<thick*.85; const mouth=md && y>R*.12 && Math.abs(x)<R*.53;
    t.dark[r][c]=ring||eye||mouth; t.weight[r][c]=(ring||eye||mouth)?1:.55;
  }
  t.label='center smiley'; state.target=t; renderAll();
}
